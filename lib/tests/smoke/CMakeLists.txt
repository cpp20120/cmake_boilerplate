foreach(_library IN ITEMS library1 library2)
  foreach(_kind IN ITEMS shared static)
    if(TARGET ${_library}_${_kind})
      set(_test "${_library}_${_kind}_smoke")
      boilerplate_add_test(${_test} SOURCES smoke.cpp LIBRARIES ${_library}_${_kind})
      if(_library STREQUAL "library1")
        target_compile_definitions(${_test} PRIVATE SAMPLE_NAMESPACE=lib1)
      else()
        target_compile_definitions(${_test} PRIVATE SAMPLE_NAMESPACE=lib2)
      endif()
      boilerplate_pgo_workload(${_test})
    endif()
  endforeach()
endforeach()
