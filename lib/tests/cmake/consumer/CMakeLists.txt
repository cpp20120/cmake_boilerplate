cmake_minimum_required(VERSION 3.26)
project(InstalledLibraryConsumer LANGUAGES CXX)
find_package(sample_math 2 CONFIG REQUIRED)
if(NOT TARGET sample_math::sample_math)
  message(FATAL_ERROR "Missing preferred package target")
endif()
enable_testing()
foreach(kind IN ITEMS shared static)
  add_executable(consume_${kind} ../project/main.cpp)
  target_link_libraries(consume_${kind} PRIVATE sample_math::${kind})
  add_test(NAME consume_${kind} COMMAND consume_${kind})
endforeach()
