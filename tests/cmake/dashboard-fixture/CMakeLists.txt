cmake_minimum_required(VERSION 3.26)
project(DashboardFixture LANGUAGES CXX)
include("${BOILERPLATE_ROOT}/lib/cmake/testing/Dashboard.cmake")
boilerplate_dashboard_prepare_llvm_coverage()
include(CTest)
include("${BOILERPLATE_ROOT}/lib/cmake/project/ProjectCapabilities.cmake")
if(DASHBOARD_FIXTURE_DISABLED)
  boilerplate_project(CAPABILITIES testing)
else()
  boilerplate_project(CAPABILITIES dashboard)
endif()
add_executable(dashboard_fixture main.cpp)
if(BOILERPLATE_COVERAGE)
  target_compile_options(dashboard_fixture PRIVATE --coverage -O0 -g)
  target_link_options(dashboard_fixture PRIVATE --coverage)
endif()
if(BOILERPLATE_LLVM_SOURCE_COVERAGE)
  target_compile_options(dashboard_fixture PRIVATE -fprofile-instr-generate -fcoverage-mapping -O0 -g)
  target_link_options(dashboard_fixture PRIVATE -fprofile-instr-generate -fcoverage-mapping)
endif()
add_test(NAME dashboard_pass COMMAND dashboard_fixture pass)
set_tests_properties(dashboard_pass PROPERTIES LABELS smoke TIMEOUT 10)
if(DASHBOARD_FIXTURE_FAIL)
  add_test(NAME dashboard_fail COMMAND dashboard_fixture fail)
  set_tests_properties(dashboard_fail PROPERTIES LABELS smoke TIMEOUT 10)
endif()
boilerplate_finalize_project()
