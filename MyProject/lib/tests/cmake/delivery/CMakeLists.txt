cmake_minimum_required(VERSION 3.26)
project(DeliveryFixture LANGUAGES CXX)
include(CTest)
include("${BOILERPLATE_MODULE_DIR}/Boilerplate.cmake")
set(BOILERPLATE_COMPILER_CACHE none CACHE STRING "" FORCE)
boilerplate_project(CAPABILITIES project-minimal packaging)
get_filename_component(_example "${CMAKE_CURRENT_LIST_DIR}/../../../../examples/plugins" ABSOLUTE)
add_library(external SHARED IMPORTED)
add_library(external_leaf SHARED IMPORTED)
set_target_properties(external_leaf PROPERTIES IMPORTED_LOCATION "${EXTERNAL_LEAF}")
set_target_properties(external PROPERTIES IMPORTED_LOCATION "${EXTERNAL_LIBRARY}")
target_link_libraries(external INTERFACE external_leaf)
if(WIN32)
  set_property(TARGET external PROPERTY IMPORTED_IMPLIB "${EXTERNAL_IMPLIB}")
  set_property(TARGET external_leaf PROPERTY IMPORTED_IMPLIB "${EXTERNAL_LEAF_IMPLIB}")
endif()
add_library(owned SHARED owned.cpp)
# A development archive in the same libdir must not be treated as an ELF DSO.
add_library(unrelated_archive STATIC empty.cpp)
install(TARGETS unrelated_archive ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}" COMPONENT Development)
target_link_libraries(owned PRIVATE external)
set_target_properties(owned PROPERTIES BUILD_RPATH_USE_ORIGIN ON)
boilerplate_add_plugin(example_plugin SOURCES "${_example}/plugin.cpp" LIBRARIES owned POLICIES minimal)
target_compile_definitions(example_plugin PRIVATE EXAMPLE_PLUGIN_DEPENDENCY)
boilerplate_add_executable(plugin_host SOURCES "${_example}/host.cpp" LIBRARIES ${CMAKE_DL_LIBS} POLICIES minimal)
boilerplate_install_application(plugin_host LIBRARIES owned PLUGINS example_plugin RESOURCE_DIRS "${_example}/resources")
boilerplate_add_plugin(wrong_plugin SOURCES "${_example}/plugin.cpp" POLICIES minimal)
target_include_directories(wrong_plugin PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/example_plugin")
target_compile_definitions(wrong_plugin PRIVATE EXAMPLE_PLUGIN_VERSION=2 EXAMPLE_PLUGIN_EXPORTS)
boilerplate_add_plugin(empty_plugin SOURCES empty.cpp POLICIES minimal)
foreach(_target IN ITEMS example_plugin wrong_plugin plugin_host)
  target_include_directories(${_target} PRIVATE "${_example}")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/paths-$<CONFIG>.cmake" CONTENT
  "set(host [==[$<TARGET_FILE:plugin_host>]==])\nset(plugin [==[$<TARGET_FILE:example_plugin>]==])\nset(wrong [==[$<TARGET_FILE:wrong_plugin>]==])\nset(empty [==[$<TARGET_FILE:empty_plugin>]==])\nset(installed_host [==[${CMAKE_INSTALL_BINDIR}/$<TARGET_FILE_NAME:plugin_host>]==])\nset(installed_plugin [==[$<TARGET_PROPERTY:plugin_host,BOILERPLATE_INSTALL_PLUGIN_DIR>/$<TARGET_FILE_NAME:example_plugin>]==])\nset(resource [==[$<TARGET_PROPERTY:plugin_host,BOILERPLATE_INSTALL_RESOURCE_DIR>/message.txt]==])\n")
boilerplate_finalize_project()
