cmake_minimum_required(VERSION 3.26)
project(CrossFixture VERSION 1.0.0 LANGUAGES CXX)
include(CTest)
include("${TOOLKIT_DIR}/Boilerplate.cmake")
if(NOT CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Fixture must configure as a cross build")
endif()
# Any configure-time target execution is a bug in this fixture/framework path.
function(try_run)
  message(FATAL_ERROR "Unexpected try_run during cross configuration")
endfunction()
set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM ONLY)
boilerplate_project_tool(_cmake "host CMake" NAMES cmake)
if(NOT _cmake STREQUAL CMAKE_COMMAND)
  message(FATAL_ERROR "Host tool resolved from target sysroot: ${_cmake}")
endif()
boilerplate_add_host_tool(host_codegen NAMES cmake)
get_target_property(_host_program host_codegen IMPORTED_LOCATION)
if(NOT _host_program STREQUAL CMAKE_COMMAND)
  message(FATAL_ERROR "Imported host codegen resolved incorrectly")
endif()
set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)
find_package(HostOnly CONFIG QUIET)
if(HostOnly_FOUND)
  message(FATAL_ERROR "find_package leaked a host prefix")
endif()
find_package(TargetOnly CONFIG REQUIRED)
boilerplate_project(CAPABILITIES testing packaging)
boilerplate_add_library(crossmath SOURCES math.cpp INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/include")
boilerplate_add_executable(cross_probe SOURCES main.cpp LIBRARIES crossmath::crossmath INSTALL)
if(EXPECT_EMULATOR)
  target_compile_definitions(cross_probe PRIVATE EXPECT_EMULATOR)
endif()
boilerplate_register_check(cross_probe cross_probe)
boilerplate_discover_tests(cross_probe TEST_PREFIX discovered. DISCOVERY_MODE PRE_TEST)
boilerplate_add_scenario(probe TARGET cross_probe REPEATS 1 WARMUP 0)
boilerplate_add_harness(matrix TARGET cross_probe ROUNDS 1 WARMUP_RUNS 0 PARSER json-line METRICS value INVARIANTS checksum)
boilerplate_add_executable(pgo_probe SOURCES main.cpp LIBRARIES crossmath::crossmath POLICY_OPTIONS PGO_MODE generate)
if(EXPECT_EMULATOR)
  target_compile_definitions(pgo_probe PRIVATE EXPECT_EMULATOR)
endif()
boilerplate_pgo_workload(pgo_probe)
boilerplate_add_reference(cross_reference SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/consumer")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/target-path.txt" CONTENT "$<TARGET_FILE:cross_probe>")
add_custom_target(generate_host_file ALL COMMAND host_codegen -E touch "${CMAKE_CURRENT_BINARY_DIR}/host-generated.txt")
if(TEST_NATIVE_POLICY)
  boilerplate_add_executable(bad_native SOURCES main.cpp POLICIES native LIBRARIES crossmath::crossmath)
endif()
boilerplate_finalize_project()
